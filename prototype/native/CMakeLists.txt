cmake_minimum_required(VERSION 3.22)
project(focuspilot_local LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "ARM instruction baseline; optimized builds require matching hardware")
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "Use the pinned upstream KleidiAI CPU kernels")
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../research/llama.cpp llama)
add_library(focuspilot_core STATIC core.cpp)
target_include_directories(focuspilot_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(focuspilot_core PUBLIC llama)
if(ANDROID)
  add_library(focuspilot_local SHARED jni.cpp)
  target_link_libraries(focuspilot_local PRIVATE focuspilot_core log)
  target_link_options(focuspilot_local PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(focuspilot_probe probe.cpp)
  target_link_libraries(focuspilot_probe PRIVATE focuspilot_core)
endif()
